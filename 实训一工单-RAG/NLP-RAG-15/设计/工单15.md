**1. 查询理解优化（`rag/nlp/search.py`）**

```python
# 新增：视觉引用识别与查询重写
import re

class VisualQueryRewriter:
    """识别问题中的视觉引用，增强检索查询"""
    
    FIGURE_PATTERN = re.compile(r'第?(\d+)页?[的]?图[号]?(\d+)')
    PAGE_PATTERN = re.compile(r'第(\d+)页')
    
    def rewrite(self, query: str) -> dict:
        """重写查询，提取视觉引用"""
        result = {
            'original_query': query,
            'text_query': query,
            'visual_refs': [],
            'enhanced_query': query
        }
        
        # 匹配"第11页图3"
        fig_matches = self.FIGURE_PATTERN.findall(query)
        for page, fig in fig_matches:
            result['visual_refs'].append({
                'type': 'figure',
                'page': int(page),
                'figure': int(fig),
                'es_filter': {
                    'bool': {
                        'must': [
                            {'term': {'page_num_int': int(page)}},
                            {'term': {'figure_num': int(fig)}}
                        ]
                    }
                }
            })
            # 从文本查询中移除视觉引用，避免干扰文本匹配
            result['text_query'] = result['text_query'].replace(
                f'第{page}页图{fig}', ''
            )
        
        # 构建增强查询：原始查询 + 视觉定位条件
        if result['visual_refs']:
            result['enhanced_query'] = {
                'must': {'multi_match': {'query': result['text_query'], 'fields': ['content_with_weight', 'title_tks']}},
                'filter': result['visual_refs'][0]['es_filter']
            }
        
        return result
```

**2. 多路召回融合（`rag/nlp/search.py`）**

```python
class MultiModalRetriever:
    """多路召回：文本检索 + 图像检索 + 图文联合检索"""
    
    def retrieve(self, query_rewritten: dict, kb_id: str, top_k: int = 10) -> list:
        """执行多路召回并融合"""
        results = []
        
        # 路1：文本检索（原始查询）
        text_results = self._text_search(
            query_rewritten['text_query'], 
            kb_id, 
            top_k=top_k
        )
        results.append(('text', text_results))
        
        # 路2：视觉引用精确检索（如有图号）
        if query_rewritten['visual_refs']:
            visual_results = self._visual_search(
                query_rewritten['visual_refs'][0],
                kb_id,
                top_k=top_k
            )
            results.append(('visual', visual_results))
            
            # 路3：图像描述文本检索（VLM生成的图像描述）
            img_desc_query = self._build_image_desc_query(
                query_rewritten['visual_refs'][0]
            )
            img_desc_results = self._text_search(
                img_desc_query,
                kb_id,
                top_k=top_k
            )
            results.append(('img_desc', img_desc_results))
        
        # 融合：RRF (Reciprocal Rank Fusion)
        fused = self._reciprocal_rank_fusion(results, k=60)
        return fused[:top_k]
    
    def _reciprocal_rank_fusion(self, results: list, k: int = 60) -> list:
        """RRF融合多路结果"""
        scores = {}
        for source_name, source_results in results:
            for rank, item in enumerate(source_results):
                doc_id = item['id']
                if doc_id not in scores:
                    scores[doc_id] = {'item': item, 'score': 0}
                # RRF公式
                scores[doc_id]['score'] += 1 / (k + rank + 1)
        
        # 按分数排序
        sorted_results = sorted(
            scores.values(), 
            key=lambda x: x['score'], 
            reverse=True
        )
        return [r['item'] for r in sorted_results]
```

**3. 跨模态重排序（`rag/nlp/rerank.py`）**

```python
class CrossModalReranker:
    """轻量级跨模态重排序"""
    
    def __init__(self):
        # 使用BGE-M3或类似跨模态模型
        self.reranker = FlagReranker('BAAI/bge-reranker-v2-m3')
    
    def rerank(self, query: str, candidates: list) -> list:
        """对候选chunk进行跨模态重排序"""
        pairs = []
        for cand in candidates:
            # 构建查询-文档对
            if cand.get('img_id'):  # 图像chunk
                # 使用VLM生成的图像描述作为文本代理
                doc_text = cand.get('image_description', '') or cand.get('content_with_weight', '')
            else:  # 文本chunk
                doc_text = cand.get('content_with_weight', '')
            
            pairs.append([query, doc_text])
        
        # 计算相关性分数
        scores = self.reranker.compute_score(pairs)
        
        # 排序
        for cand, score in zip(candidates, scores):
            cand['rerank_score'] = score
        
        return sorted(candidates, key=lambda x: x['rerank_score'], reverse=True)
```

**4. 提示工程优化（`rag/prompts/generator.py`）**

```python
def build_multimodal_prompt(query: str, chunks: list, dialog_history: list) -> str:
    """构建多模态提示，明确指示LLM结合图纸分析"""
    
    # 分离文本和图像chunk
    text_chunks = [c for c in chunks if not c.get('img_id')]
    image_chunks = [c for c in chunks if c.get('img_id')]
    
    prompt = f"""## 任务
基于以下检索到的专利文档片段，回答用户问题。

## 检索到的文本信息
"""
    for i, chunk in enumerate(text_chunks[:3], 1):
        prompt += f"\n### 片段{i}\n{chunk['content_with_weight']}\n"
    
    # 图像chunk特殊处理
    if image_chunks:
        prompt += """
## 检索到的技术图纸信息
"""
        for img in image_chunks[:2]:
            prompt += f"""
### 图纸：{img.get('figure_num', '未知图号')}
- 所在页码：第{img.get('page_num_int', '?')}页
- 图像描述：{img.get('image_description', '无描述')}
- 图纸中的部件编号：{img.get('detected_labels', '未识别')}
"""
    
    prompt += f"""
## 用户问题
{query}

## 回答要求
1. 如果问题涉及图纸中的部件编号或位置关系，请**优先基于图纸信息**回答
2. 分析图纸时，请按以下步骤：
   - 先确认图号
   - 定位问题中提到的部件编号
   - 描述部件之间的空间关系（左/右/上/下/内/外）
   - 追踪气流或物料流动路径
3. 如果文本片段与图纸信息冲突，以**图纸信息为准**
4. 以结构化格式输出，包含推理过程和最终答案

## 回答
"""
    return prompt
```
